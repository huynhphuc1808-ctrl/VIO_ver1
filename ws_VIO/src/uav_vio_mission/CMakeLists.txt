cmake_minimum_required(VERSION 3.16)
project(uav_vio_mission)

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()

add_compile_options(-Wall -Wextra -Wpedantic -Werror -Wno-unused-parameter)

find_package(ament_cmake REQUIRED)
find_package(eigen3_cmake_module REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_ros2_cpp REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Eigen3 REQUIRED)

add_library(mission_definition STATIC src/mission_definition.cpp)
target_include_directories(mission_definition PUBLIC include)
target_link_libraries(mission_definition Eigen3::Eigen px4_ros2_cpp::px4_ros2_cpp rclcpp::rclcpp)
ament_target_dependencies(mission_definition px4_msgs std_msgs)

add_executable(vio_mission src/vio_mission.cpp src/mission_context.cpp src/mission_actions.cpp src/mission_runtime.cpp)
target_link_libraries(vio_mission mission_definition)
ament_target_dependencies(vio_mission px4_msgs std_msgs)
install(TARGETS vio_mission DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(mission_logic_test test/test_mission_logic.cpp)
  target_include_directories(mission_logic_test PRIVATE include)
  ament_add_gtest(mission_definition_test test/test_mission_definition.cpp)
  target_link_libraries(mission_definition_test mission_definition)
  ament_add_gtest(mission_guards_test test/test_mission_guards.cpp)
  target_link_libraries(mission_guards_test mission_definition)
  ament_add_gtest(mission_file_observer_test test/test_mission_file_observer.cpp)
  target_link_libraries(mission_file_observer_test mission_definition)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME mission_analysis_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/test_mission_analysis.py)
  foreach(fixture circle circle_ccw circle_multilap two_orbits square local_goto_orbit local_takeoff_land)
    add_test(NAME validate_${fixture} COMMAND vio_mission --validate-mission ${CMAKE_CURRENT_SOURCE_DIR}/config/${fixture}.json)
  endforeach()
endif()

ament_package()
